cmake_minimum_required(VERSION 3.16)
project(offline_sync_bench LANGUAGES CXX)
enable_testing()
add_executable(sync_tests tests/test.cpp)
target_include_directories(sync_tests PRIVATE include)
set_target_properties(sync_tests PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(sync_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME sync_tests COMMAND sync_tests)
